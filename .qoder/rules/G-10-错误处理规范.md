# G-10: 错误处理规范

## 错误分类

### 业务错误

用户操作导致的可预期错误：

- 参数校验失败
- 资源不存在
- 权限不足
- 重复操作

```go
// Go 示例
var ErrUserNotFound = errors.New("用户不存在")
var ErrInvalidParameter = errors.New("参数错误")
var ErrPermissionDenied = errors.New("权限不足")
```

### 系统错误

服务内部错误，用户无法解决：

- 数据库连接失败
- 第三方服务不可用
- 内部逻辑错误

```go
var ErrInternalServer = errors.New("服务器内部错误")
var ErrDatabaseConnection = errors.New("数据库连接失败")
```

### 第三方错误

外部服务返回的错误：

- API 调用失败
- 超时
- 限流

```go
var ErrThirdPartyAPI = errors.New("第三方服务异常")
var ErrTimeout = errors.New("请求超时")
```

## 自定义错误类

### Go 实现

```go
// AppError 应用错误
type AppError struct {
    Code    int    // 错误码
    Message string // 用户友好消息
    Detail  string // 详细错误（日志用）
    Err     error  // 原始错误
}

func (e *AppError) Error() string {
    return e.Message
}

func (e *AppError) Unwrap() error {
    return e.Err
}

// 构造函数
func NewError(code int, message string) *AppError {
    return &AppError{
        Code:    code,
        Message: message,
    }
}

func NewErrorWithDetail(code int, message, detail string) *AppError {
    return &AppError{
        Code:    code,
        Message: message,
        Detail:  detail,
    }
}

func WrapError(code int, message string, err error) *AppError {
    return &AppError{
        Code:    code,
        Message: message,
        Detail:  err.Error(),
        Err:     err,
    }
}
```

### TypeScript 实现

```typescript
class AppError extends Error {
  constructor(
    public code: number,
    public message: string,
    public detail?: string,
    public originalError?: Error,
  ) {
    super(message);
    this.name = "AppError";
  }
}

// 便捷函数
export const createError = (code: number, message: string) =>
  new AppError(code, message);

export const wrapError = (code: number, message: string, err: Error) =>
  new AppError(code, message, err.message, err);
```

### Python 实现

```python
class AppError(Exception):
    def __init__(self, code: int, message: str, detail: str = ""):
        self.code = code
        self.message = message
        self.detail = detail
        super().__init__(message)


class NotFoundError(AppError):
    def __init__(self, resource: str, resource_id: str):
        super().__init__(code=3001, message=f"{resource} 不存在")


def raise_if_none(value, code: int, message: str):
    if value is None:
        raise AppError(code, message)
    return value
```

## 错误处理模式

### Go 错误处理

```go
// 立即处理
user, err := repo.GetUser(id)
if err != nil {
    if errors.Is(err, gorm.ErrRecordNotFound) {
        return nil, NewError(CodeResourceNotFound, "用户不存在")
    }
    logger.Error("获取用户失败", zap.Error(err))
    return nil, NewError(CodeDatabaseError, "查询失败")
}

// 包装错误
if err := service.DoSomething(); err != nil {
    return WrapError(CodeBusinessRuleViolation, "操作失败", err)
}

// 错误链
if err := step1(); err != nil {
    return fmt.Errorf("step1 failed: %w", err)
}
if err := step2(); err != nil {
    return fmt.Errorf("step2 failed: %w", err)
}
```

### TypeScript 错误处理

```typescript
// async/await
try {
  const user = await fetchUser(id);
} catch (error) {
  if (error instanceof AppError) {
    toast.error(error.message);
  } else {
    toast.error("未知错误");
    console.error("Unexpected error:", error);
  }
}

// 错误边界（Vue）
const handleError = (error: unknown) => {
  if (axios.isAxiosError(error)) {
    const code = error.response?.data?.code || CodeInternalError;
    const message = error.response?.data?.message || "请求失败";
    toast.error(message);
    return;
  }
  toast.error("操作失败");
  console.error(error);
};
```

## 用户友好提示

### 原则

1. **清晰**：用户能看懂发生了什么
2. **可操作**：告诉用户下一步怎么做
3. **不暴露细节**：不显示技术细节（如 SQL 错误）

### 示例

```go
// 不好
"sql: no rows in result set"
"Error 1146: Table 'db.users' doesn't exist"

// 好
"用户不存在，请检查输入是否正确"
"系统繁忙，请稍后重试"
```

### 按场景提示

| 场景       | 用户提示               | 日志记录               |
| ---------- | ---------------------- | ---------------------- |
| 参数错误   | "输入信息有误，请检查" | 详细参数和校验失败原因 |
| 资源不存在 | "请求的内容不存在"     | 资源类型和 ID          |
| 权限不足   | "您没有权限执行此操作" | 用户 ID 和尝试的操作   |
| 系统错误   | "系统繁忙，请稍后重试" | 完整错误堆栈           |
| 第三方错误 | "服务暂时不可用"       | 第三方返回的原始错误   |

## 全局错误处理器

### Go Gin 中间件

```go
func ErrorHandler() gin.HandlerFunc {
    return func(c *gin.Context) {
        c.Next()

        if len(c.Errors) > 0 {
            err := c.Errors.Last().Err

            // 处理自定义错误
            var appErr *AppError
            if errors.As(err, &appErr) {
                c.JSON(http.StatusOK, Response{
                    Code:    appErr.Code,
                    Message: appErr.Message,
                })
                // 记录详细错误
                logger.Error("业务错误",
                    zap.Int("code", appErr.Code),
                    zap.String("message", appErr.Message),
                    zap.String("detail", appErr.Detail),
                )
                return
            }

            // 未知错误
            logger.Error("系统错误", zap.Error(err))
            c.JSON(http.StatusOK, Response{
                Code:    CodeInternalError,
                Message: "系统繁忙，请稍后重试",
            })
        }
    }
}
```

### Vue 全局处理

```typescript
// 请求拦截器
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response) {
      const { code, message } = error.response.data;
      toast.error(message || "请求失败");
      return Promise.reject(new AppError(code, message));
    }
    toast.error("网络错误");
    return Promise.reject(error);
  },
);

// 全局错误处理
app.config.errorHandler = (err, vm, info) => {
  console.error("Global error:", err);
  console.error("Component:", vm);
  console.error("Info:", info);
  // 上报错误
};
```

## 错误日志规范

### 日志内容

```go
// 必须包含
logger.Error("操作失败",
    zap.String("trace_id", traceID),      // 追踪 ID
    zap.String("user_id", userID),        // 用户 ID
    zap.String("operation", "create_campaign"), // 操作
    zap.Error(err),                        // 错误
    zap.Any("params", params),             // 参数
)
```

### 日志级别

| 级别  | 使用场景             |
| ----- | -------------------- |
| DEBUG | 详细调试信息         |
| INFO  | 正常业务日志         |
| WARN  | 可恢复异常（如重试） |
| ERROR | 需要处理的错误       |
| FATAL | 服务无法继续运行     |

## 错误处理检查清单

```
□ 所有错误都被捕获和处理
□ 没有裸的 panic/recover
□ 用户看到的是友好提示，不是技术错误
□ 日志记录了足够的上下文信息
□ 错误码定义清晰，不重复
□ 敏感信息不出现在错误消息中
□ 第三方错误被包装，不直接暴露
□ 生产环境不返回堆栈信息
```
